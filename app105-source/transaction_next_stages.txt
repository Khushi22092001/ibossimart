Payment Allocation    /* The bridge itself, one row per payment x bill. This is the only place                                      
                         in the dashboard where the many-to-many relationship is shown at its                                       
                         own grain - everywhere else it is pre-aggregated before joining, or                                        
                         the outstanding figure would duplicate. On the AP side the BILL is                                         
                         the credit line (cv) and the PAYMENT is the debit line (dv) - the                                          
                         exact mirror of the receivables page. */                                                                   
                      With Asof As (Select to_date(:P923_TODATE,'DD-MM-RRRR') D From dual)                                          
                      Select cv.VoucherNo                                 As BILL_VOUCHER,                                          
                             cv.Tno                                       As BILL_TNO,                                              
                             coalesce(pb.PartyBillNo, ao.BillNo, cv.VoucherNo) As BILL_REFERENCE,                                   
                             cv.VoucherDate                                As BILL_DATE,                                            
                             dv.VoucherNo                                  As RECEIPT_VOUCHER,                                      
                             dv.Tno                                        As RECEIPT_TNO,                                          
                             dv.VoucherDate                                As RECEIPT_DATE,                                         
                             Round(a.Amount,2)                             As ALLOCATED,                                            
                             dv.VoucherDate - cv.VoucherDate               As DAYS_TO_SETTLE,                                       
                             nvl(cl.LocationName, cd.LocationCode) As BILL_LOCATION,                                                
                             nvl(dl.LocationName, dd.LocationCode) As RECEIPT_LOCATION,                                             
                             Case When dd.LocationCode <> cd.LocationCode                                                           
                                       Then '<span class="ds-apchip ds-apchip--warn">Cross-location</span>'                         
                                  Else '<span class="ds-apchip ds-apchip--pass">Same location</span>' End As SETTLEMENT_TYPE        
                        From DrCrAllocation a                                                                                       
                        Join Voucher dv On dv.Tno = a.DrVoucherTno                                                                  
                        Join Voucher cv On cv.Tno = a.CrVoucherTno                                                                  
                        Join VoucherDetail cd On cd.Tno = a.CrVoucherTno And cd.Sno = a.CrVoucherSno                                
                        Left Join VoucherDetail dd On dd.Tno = a.DrVoucherTno And dd.Sno = a.DrVoucherSno                           
                        Left Join PBPass pp On cv.ModuleCode = 'PBPASS' And pp.Tno = cv.ModuleTno                                   
                        Left Join PurchaseBill pb On pb.Tno = pp.PurchaseBillTno                                                    
                        Left Join AccountOpening ao On cv.VoucherNo = 'OPENING' And ao.Tno = cd.ModuleTno                           
                        Left Join Location dl On dl.LocationCode = dd.LocationCode                                                  
                        Left Join Location cl On cl.LocationCode = cd.LocationCode                                                  
                        Cross Join Asof                                                                                             
                       Where cd.AccountCode = :P923_PARTY                                                                           
                         And dv.VoucherDate <= Asof.D And cv.VoucherDate <= Asof.D                                                  
Open Items            With Asof As (Select to_date(:P923_TODATE,'DD-MM-RRRR') D From dual),

                              Group By a.DrVoucherTno, a.DrVoucherSno

                               From DrCrAllocation a, Voucher dv, Voucher cv, Asof

                              Group By a.CrVoucherTno, a.CrVoucherSno),

                      Select d.Tno                                          As VOUCHER_TNO,                                         
                             /* Sno is what makes a bill addressable. A voucher can carry many

                             Round(Abs(d.Amount),2)                         As ORIGINAL_AMOUNT,                                     
                             Round(Abs(nvl(al.Amt,0)),2)                     As ALLOCATED,                                          
                             Round(Case When (d.Amount - nvl(al.Amt,0)) > 0                                                         
                                        Then (d.Amount - nvl(al.Amt,0)) Else 0 End,2) As OUTSTANDING,                               
                             Round(Case When (d.Amount - nvl(al.Amt,0)) < 0                                                         
                                        Then  (d.Amount - nvl(al.Amt,0)) Else 0 End,2) As CREDIT_OPEN,                              
                             Case When coalesce(pt.DueDt, jt.DueDt, ao.BillDueDate) Is Not Null                                     
                                  Then Greatest((Select D From Asof) - coalesce(pt.DueDt, jt.DueDt, ao.BillDueDate),0)              
                             End                                            As DAYS_OVERDUE,                                        
                             nvl(l.LocationName, d.LocationCode) As LOCATION,                                                       
                             substr(d.Narration,1,300)    As NARRATION                                                              
                        From VoucherDetail d                                                                                        
                        Join Voucher v On v.Tno = d.Tno                                                                             
                        Left Join Al al On al.Tno = d.Tno And al.Sno = d.Sno                                                        
                        Left Join (Select pp.Tno Tno,                                                                               
                                                 coalesce(Max(pb.PartyBillDate), Max(pb.PurchaseBillDate)) DocDt,

                                                   + coalesce(Max(po.CreditDays), Max(pty.CreditDays)) DueDt                        
                                            From PBPass pp                                                                          
                                            Join PurchaseBill pb On pb.Tno = pp.PurchaseBillTno                                     
                                            Left Join PurchaseBillDetail pbd On pbd.Tno = pb.Tno                                    
                                            Left Join PurchaseOrder po On po.Tno = pbd.PurchaseOrderTno                             
                                            Left Join Party pty On pty.PartyCode = pb.PartyCode                                     
                                           Group By pp.Tno) pt On v.ModuleCode = 'PBPASS' And pt.Tno = v.ModuleTno                  
                               Left Join (Select jp.Tno Tno,                                                                        
                                                 coalesce(Max(jb.PartyBillDate), Max(jb.JobBillDate)) DocDt,                        
                                                 Max(jb.JobBillDate) + nvl(Max(jo.CreditDays),0) DueDt                              
                                            From JBPass jp                                                                          
                                            Join JobBill jb On jb.Tno = jp.JobBillTno                                               
                                            Left Join JobOrder jo On jo.Tno = jb.JobOrderTno                                        
                                           Group By jp.Tno) jt On v.ModuleCode = 'JBPASS' And jt.Tno = v.ModuleTno

                        Cross Join Asof                                                                                             
                       Where d.AccountCode = :P923_PARTY

Payment History       /* A payment line is a DEBIT - its settlements sit on the Dr side of                                          
                         the bridge, the mirror of the receivables page. */                                                         
                      With Asof As (Select to_date(:P923_TODATE,'DD-MM-RRRR') D From dual),                                         
                           Alc As (                                                                                                 
                             Select a.DrVoucherTno Tno, a.DrVoucherSno Sno,                                                         
                                    Sum(a.Amount) Amt, Count(*) N                                                                   
                               From DrCrAllocation a, Voucher dv, Voucher cv, Asof                                                  
                              Where dv.Tno = a.DrVoucherTno And cv.Tno = a.CrVoucherTno                                             
                                And dv.VoucherDate <= Asof.D And cv.VoucherDate <= Asof.D

                             v.VoucherNo                                    As RECEIPT_NO,                                          
                             d.VoucherDate                                  As RECEIPT_DATE,                                        
                             Round(-d.Amount,2)                             As RECEIPT_AMOUNT,                                      
                             Round(nvl(al.Amt,0),2)                         As APPLIED,                                             
                             Round(-d.Amount - nvl(al.Amt,0),2)             As UNAPPLIED,                                           
                             nvl(al.N,0)                                    As BILLS_SETTLED,                                       
                             Case When nvl(al.N,0) = 0                                                                              
                                       Then '<span class="ds-apchip ds-apchip--fail">Unapplied</span>'                              
                                  When -d.Amount - nvl(al.Amt,0) > 0.005                                                            
                                       Then '<span class="ds-apchip ds-apchip--warn">Partly applied</span>'                         
                                  Else '<span class="ds-apchip ds-apchip--pass">Fully applied</span>' End As STATUS,                
                             v.MoneyTransferReferenceNo   As BANK_REFERENCE,                                                        
                             nvl(l.LocationName, d.LocationCode) As LOCATION,                                                       
                             substr(d.Narration,1,300)    As NARRATION                                                              
                        From VoucherDetail d

Vendor Statement      /* A true statement of account: every line on the vendor, in date                                             
                         order, with the running balance carried down. The running total is a                                       
                         window function over the SAME rows the reader sees, so the closing                                         
                         figure at the bottom equals the Net Exposure KPI at the top by                                             
                         construction rather than by coincidence. */                                                                
                      Select d.VoucherDate                                As TRANSACTION_DATE,                                      
                             v.VoucherNo                                  As VOUCHER_NO,                                            
                             d.Tno                                        As VOUCHER_TNO,                                           
                             v.DocTypeCode                                As VOUCHER_TYPE,                                          
                             coalesce(pb.PartyBillNo, ao.BillNo, v.VoucherNo) As REFERENCE,                                         
                             Round(Case When d.Amount < 0 Then -d.Amount End,2) As DEBIT,                                           
                             Round(Case When d.Amount > 0 Then  d.Amount End,2) As CREDIT,                                          
                             Round(Sum(d.Amount) Over (Order By d.VoucherDate, d.Tno, d.Sno                                         
                                                        Rows Between Unbounded Preceding And Current Row),2) As RUNNING_BALANCE,    
                             nvl(l.LocationName, d.LocationCode) As LOCATION,                                                       
                             substr(d.Narration,1,250)  As NARRATION                                                                
                        From VoucherDetail d                                                                                        
                        Join Voucher v On v.Tno = d.Tno                                                                             
                        Left Join PBPass pp On v.ModuleCode = 'PBPASS' And pp.Tno = v.ModuleTno                                     
                        Left Join PurchaseBill pb On pb.Tno = pp.PurchaseBillTno                                                    
                        Left Join AccountOpening ao On v.VoucherNo = 'OPENING' And ao.Tno = d.ModuleTno                             
                        Left Join Location l On l.LocationCode = d.LocationCode                                                     
                       Where d.AccountCode = :P923_PARTY                                                                            
                         And d.VoucherDate <= to_date(:P923_TODATE,'DD-MM-RRRR')                                                    
                         And (cast(null as varchar2(100)) Is Null                                                                   
                              Or cast(null as varchar2(100)) = cast(null as varchar2(100)))                                         
                         And (null    Is Null Or cast(null as varchar2(100))        = null)                                         
                         And (:P923_COMPANY  Is Null Or d.CompanyCode  = :P923_COMPANY)                                             
                         And (:P923_LOCATION Is Null Or d.LocationCode = :P923_LOCATION)
