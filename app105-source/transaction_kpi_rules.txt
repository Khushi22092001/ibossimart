V_LATEST_ACTIVE_SALESORDERS    WITH REVISION_TREE AS (                                                                                                                                                          
                                   SELECT                                                                                                                                                                       
                                       TNO AS CURRENT_SO_TNO,                                                                                                                                                   
                                       SalesOrderNo AS CURRENT_SO_NO,                                                                                                                                           
                                       CONNECT_BY_ROOT TNO AS ROOT_SO_TNO,                                                                                                                                      
                                       CONNECT_BY_ROOT SalesOrderNo AS ROOT_SO_NO                                                                                                                               
                                   FROM SalesOrder                                                                                                                                                              
                                   START WITH REVISIONSALESORDERTNO IS NULL                                                                                                                                     
                                   CONNECT BY PRIOR TNO = REVISIONSALESORDERTNO                                                                                                                                 
                               ),                                                                                                                                                                               
                               root_so_quantities AS (                                                                                                                                                          
                                   -- Fetch original baseline transaction targets                                                                                                                               
                                   SELECT sod.TNO, sod.ITEMCODE, sod.ITEMSPECIFICATIONCODE, NVL(sod.Quantity1, 0) AS BASE_SO_QTY, NVL(sod.RATE, 0) AS BASE_SO_RATE, NVL(sod.TotalAmount,0) as BASE_SO_AMOUNT    
                                   FROM SalesOrderDetail sod                                                                                                                                                    
                               )                                                                                                                                                                                
                               SELECT DISTINCT                                                                                                                                                                  
                                   so.TNO,                                                                                                                                                                      
                                   REGEXP_REPLACE(so.SalesOrderNo, '.*/0*([0-9]+)', '0\1') AS SO_No_Clean,                                                                                                      
                                   so.SalesOrderDate,                                                                                                                                                           
                                   so.SalesOrderNo,                                                                                                                                                             
                                   so.PartyPONO,                                                                                                                                                                
                                   so.PartyPODate,                                                                                                                                                              
                                   so.PartyCode,                                                                                                                                                                
                                   so.Agentcode,                                                                                                                                                                
                                   so.SALESEXECUTIVECODE,                                                                                                                                                       
                                   so.REMARK,                                                                                                                                                                   
                                   sod.ITEMCODE,                                                                                                                                                                
                                   sod.ITEMSPECIFICATIONCODE,                                                                                                                                                   
                                   NVL(sod.Quantity1, 0) AS CURRENT_SO_QTY,                                                                                                                                     
                                   sod.Rate,                                                                                                                                                                    
                                   sod.Amount,                                                                                                                                                                  
                                   rt.ROOT_SO_TNO,                                                                                                                                                              
                                   rt.ROOT_SO_NO,                                                                                                                                                               
                                   NVL(rsq.BASE_SO_QTY, 0) AS BASE_SO_QTY,                                                                                                                                      
                                   NVL(rsq.BASE_SO_RATE, 0) AS BASE_SO_RATE,                                                                                                                                    
                                   NVL(rsq.BASE_SO_AMOUNT,0) AS BASE_SO_AMOUNT                                                                                                                                  
                               FROM SalesOrder so                                                                                                                                                               
                               JOIN REVISION_TREE rt ON so.TNO = rt.CURRENT_SO_TNO                                                                                                                              
                               JOIN SalesOrderDetail sod ON so.TNo = sod.TNO                                                                                                                                    
                               LEFT JOIN root_so_quantities rsq ON rt.ROOT_SO_TNO = rsq.TNO                                                                                                                     
                                   AND sod.ItemCode = rsq.ItemCode                                                                                                                                              
                                   AND sod.ItemSpecificationCode = rsq.ItemSpecificationCode                                                                                                                    
                               WHERE EXISTS (                                                                                                                                                                   
                                   SELECT 1 FROM DOCUMENTSTATUSDETAIL dsd                                                                                                                                       
                                   WHERE dsd.MODULECODE = 'SALESORDER'                                                                                                                                          
                                     AND dsd.MODULETNO = so.TNO                                                                                                                                                 
                                     AND dsd.DOCUMENTSTATUSCODE = 'ACTIVE'                                                                                                                                      
                               )                                                                                                                                                                                
                               AND NOT EXISTS (                                                                                                                                                                 
                                   SELECT 1 FROM SalesOrder child_so                                                                                                                                            
                                   WHERE child_so.REVISIONSALESORDERTNO = so.TNO                                                                                                                                
                               )
